   241→   241→function saveSearchHistory(q) {
   242→   242→  if (!q) return;
   243→   243→  try {
   244→   244→    const key = 'searchHistory';
   245→   245→    const existing = JSON.parse(localStorage.getItem(key) || '[]');
   246→   246→    const newArr = [q, ...existing.filter(s => s !== q)].slice(0, 5);
   247→   247→    localStorage.setItem(key, JSON.stringify(newArr));
   248→   248→  } catch (e) {
   249→   249→    console.warn('saveSearchHistory error:', e);
   250→   250→  }
   251→   251→}

   260→   260→  el.classList.remove('hidden');

   262→   262→  el.querySelectorAll('.suggestion-item').forEach(node => {

   266→   266→      if (sb) { sb.value = val; handleSearch({target: {value: val}}); }

   445→   445→ */
   446→   446→function togglePanel(panel, btn) {
   447→   447→  if (!panel || !btn) return;
   448→   448→  const expanded = btn.getAttribute('aria-expanded') === 'true';
   449→   449→  if (expanded) {
   450→   450→    panel.classList.add('hidden');
   451→   451→    panel.setAttribute('aria-hidden', 'true');
   452→   452→    btn.setAttribute('aria-expanded', 'false');
   453→   453→    btn.classList.remove('active');
   454→   454→  } else {
   455→   455→    panel.classList.remove('hidden');
   456→   456→    panel.setAttribute('aria-hidden', 'false');
   457→   457→    btn.setAttribute('aria-expanded', 'true');
   458→   458→    btn.classList.add('active');
   459→   459→    // Show backdrop
   460→   460→    const bd = document.getElementById('panelBackdrop');
   461→   461→    if (bd) bd.classList.remove('hidden');
   462→   462→  }
   463→   463→}

   499→   499→function closePanelsOnOutsideClick(e) {
   500→   500→  // If click is not within panel or corresponding toggle buttons, hide all panels
   501→   501→  const panels = ['searchSection', 'advancedFilters'];
   502→   502→  panels.forEach(id => {
   503→   503→    const panel = document.getElementById(id);
   504→   504→    const toggleBtn = document.getElementById(getToggleForPanelId(id));
   505→   505→    if (!panel || !toggleBtn) return;

   663→   663→function showToast(message, type = 'info', duration = 5000) {
   664→   664→  const toastContainer = document.getElementById('toastContainer');
   665→   665→  if (!toastContainer) return;
   666→   666→
   667→   667→  // Check if notifications are enabled
   668→   668→  const notificationsEnabled = localStorage.getItem('toastNotifications') !== 'false';
   669→   669→  if (!notificationsEnabled) return;
   670→   670→
   671→   671→  const toast = document.createElement('div');
   672→   672→  toast.className = `toast ${type}`;
   673→   673→  
   674→   674→  const icon = type === 'success' ? 'fa-check-circle' : 
   675→   675→               type === 'error' ? 'fa-exclamation-circle' :
   676→   676→               type === 'warning' ? 'fa-exclamation-triangle' : 'fa-info-circle';
   677→   677→  
   678→   678→  toast.innerHTML = `
   679→   679→    <i class="fas ${icon}"></i>
   680→   680→    <span>${message}</span>
   681→   681→    <button class="toast-close" onclick="this.parentElement.remove()">
   682→   682→      <i class="fas fa-times"></i>
   683→   683→    </button>
   684→   684→  `;
   685→   685→
   686→   686→  toastContainer.appendChild(toast);
   687→   687→
   688→   688→  // Auto remove after duration
   689→   689→  setTimeout(() => {
   690→   690→    if (toast.parentNode) {
   691→   691→      toast.style.animation = 'slideIn 0.3s ease reverse';
   692→   692→      setTimeout(() => {
   693→   693→        if (toast.parentNode) {
   694→   694→          toast.parentNode.removeChild(toast);
   695→   695→        }
   696→   696→      }, 300);
   697→   697→    }
   698→   698→  }, duration);
   699→   699→}

  1607→  1607→/**

  1609→  1609→ * @param {number} index - Current index of the website in the array
  1610→  1610→ * @param {string} direction - 'up' or 'down'
  1611→  1611→ */
  1612→  1612→function moveWebsite(index, direction) {
  1613→  1613→  // Get current editing actress from the actresses array
  1614→  1614→  if (!editingSlug) {
  1615→  1615→    showToast('Error: No actress is being edited', 'error');
  1616→  1616→    return;
  1617→  1617→  }

  1619→  1619→  const actress = actresses.find(a => a.slug === editingSlug);
  1620→  1620→  if (!actress || !actress.websites) {
  1621→  1621→    showToast('Error: Could not find websites to reorder', 'error');
  1622→  1622→    return;
  1623→  1623→  }
  1624→  1624→  
  1625→  1625→  // Make sure index is within bounds
  1626→  1626→  if (!Number.isInteger(index) || index < 0 || index >= actress.websites.length) {
  1627→  1627→    console.warn('Invalid index:', index, 'Total websites:', actress.websites.length);
  1628→  1628→    showToast('Error: Invalid website index', 'error');
  1629→  1629→    return;
  1630→  1630→  }
  1631→  1631→  
  1632→  1632→  // Determine the swap target index
  1633→  1633→  let targetIndex = -1;
  1634→  1634→  if (direction === 'up' && index > 0) {
  1635→  1635→    targetIndex = index - 1;
  1636→  1636→  } else if (direction === 'down' && index < actress.websites.length - 1) {
  1637→  1637→    targetIndex = index + 1;
  1638→  1638→  } else {
  1639→  1639→    showToast(`Cannot move ${direction} - already at ${direction === 'up' ? 'top' : 'bottom'}`, 'info');
  1640→  1640→    return;
  1641→  1641→  }
  1642→  1642→  
  1643→  1643→  // Swap the websites in the array
  1644→  1644→  [actress.websites[index], actress.websites[targetIndex]] = 
  1645→  1645→  [actress.websites[targetIndex], actress.websites[index]];
  1646→  1646→  
  1647→  1647→  // Re-populate the form to reflect the new order
  1648→  1648→  populateFormForEditing(actress);
  1649→  1649→  
  1650→  1650→  console.log(`Website moved ${direction}: old index ${index}, new index ${targetIndex}`);
  1651→  1651→}
