   241→function saveSearchHistory(q) {
   242→  if (!q) return;
   243→  try {
   244→    const key = 'searchHistory';
   245→    const existing = JSON.parse(localStorage.getItem(key) || '[]');
   246→    const newArr = [q, ...existing.filter(s => s !== q)].slice(0, 5);
   247→    localStorage.setItem(key, JSON.stringify(newArr));
   248→  } catch (e) {
   249→    console.warn('saveSearchHistory error:', e);
   250→  }
   251→}
   252→
   253→function showSearchSuggestions() {
   254→  const el = document.getElementById('searchSuggestions');
   255→  if (!el) return;
   256→  const key = 'searchHistory';
   257→  const items = JSON.parse(localStorage.getItem(key) || '[]');
   258→  if (!items.length) { el.classList.add('hidden'); return; }
   259→  el.innerHTML = items.map(s => `<div class="suggestion-item" data-value="${s}">${s}</div>`).join('');
   260→  el.classList.remove('hidden');
   261→  // Attach click handlers to suggestions
   262→  el.querySelectorAll('.suggestion-item').forEach(node => {
   263→    node.addEventListener('click', () => {
   264→      const val = node.getAttribute('data-value');
   265→      const sb = document.getElementById('searchBar');
   266→      if (sb) { sb.value = val; handleSearch({target: {value: val}}); }
   267→      hideSearchSuggestions();
   268→    });
   269→  });
   270→}

   443→/**
   444→ * Helper to toggle panels and set aria states
   445→ */
   446→function togglePanel(panel, btn) {
   447→  if (!panel || !btn) return;
   448→  const expanded = btn.getAttribute('aria-expanded') === 'true';
   449→  if (expanded) {
   450→    panel.classList.add('hidden');
   451→    panel.setAttribute('aria-hidden', 'true');
   452→    btn.setAttribute('aria-expanded', 'false');
   453→    btn.classList.remove('active');
   454→  } else {
   455→    panel.classList.remove('hidden');
   456→    panel.setAttribute('aria-hidden', 'false');
   457→    btn.setAttribute('aria-expanded', 'true');
   458→    btn.classList.add('active');
   459→    // Show backdrop
   460→    const bd = document.getElementById('panelBackdrop');
   461→    if (bd) bd.classList.remove('hidden');
   462→  }
   463→}

   465→/**
   466→ * Map panel id to toggle button id. Returns null if there is no dedicated toggle button.
   467→ */
   468→function getToggleForPanelId(id) {
   469→  if (!id) return null;
   470→  if (id === 'searchSection') return 'searchToggle';
   471→  if (id === 'advancedFilters') return 'filtersToggle';
   472→  // viewModeContainer is inline so it has no dedicated header toggle now
   473→  return null;
   474→}

   499→function closePanelsOnOutsideClick(e) {
   500→  // If click is not within panel or corresponding toggle buttons, hide all panels
   501→  const panels = ['searchSection', 'advancedFilters'];
   502→  panels.forEach(id => {
   503→    const panel = document.getElementById(id);
   504→    const toggleBtn = document.getElementById(getToggleForPanelId(id));
   505→    if (!panel || !toggleBtn) return;
   506→    if (!panel.classList.contains('hidden')) {
   507→      const inPanel = panel.contains(e.target) || toggleBtn.contains(e.target);
   508→      if (!inPanel) {
   509→        panel.classList.add('hidden');
   510→        panel.setAttribute('aria-hidden', 'true');
   511→        toggleBtn.setAttribute('aria-expanded', 'false');
   512→        toggleBtn.classList.remove('active');
   513→      }
   514→    }
   515→    // If all panels hidden, hide backdrop
   516→    const bd = document.getElementById('panelBackdrop');
   517→    if (bd) {
   518→      const sEl = document.getElementById('searchSection');
   519→      const aEl = document.getElementById('advancedFilters');
   520→      // viewModeContainer is inline, not overlay — only consider overlay panels
   521→      const any = (sEl && !sEl.classList.contains('hidden')) || (aEl && !aEl.classList.contains('hidden'));
   522→      if (!any) bd.classList.add('hidden');
   523→    }
   524→  });
   525→}

  1177→/**
  1178→ * Enhanced Search with debouncing
  1179→ */
  1180→function handleSearch(e) {
  1181→  const query = e.target.value;
  1182→  
  1183→  // Clear previous timeout
  1184→  if (searchTimeout) {
  1185→    clearTimeout(searchTimeout);
  1186→  }
  1187→  
  1188→  // Set new timeout for debouncing
  1189→  searchTimeout = setTimeout(() => {
  1190→    const filtered = filterActresses(query, currentCategory);
  1191→    displayActresses(filtered);
  1192→    updateSearchResultsCount(filtered);
  1193→    searchTimeout = null;
  1194→  }, 300);
  1195→}

  1197→/**
  1198→ * Filter actresses based on query and category
  1199→ */
  1200→function filterActresses(query, category) {
  1201→  let filtered = actresses;
  1202→  
  1203→  if (category !== 'all') {
  1204→    filtered = filtered.filter(a => a.category === category);
  1205→  }
  1206→  
  1207→  const q = String(query || '').trim().toLowerCase();
  1208→  if (q) {
  1209→    filtered = filtered.filter(a => 
  1210→      a.name.toLowerCase().includes(q) ||
  1211→      a.category.toLowerCase().includes(q) ||
  1212→      (a.tags && a.tags.some(tag => tag.toLowerCase().includes(q)))
  1213→    );
  1214→  }
  1215→  
  1216→  return filtered;
  1217→}

  1219→/**
  1220→ * Sort actresses based on criteria
  1221→ */
  1222→function sortActresses(list, sortBy) {
  1223→  return [...list].sort((a, b) => {
  1224→    switch (sortBy) {
  1225→      case 'name-desc':
  1226→        return b.name.localeCompare(a.name);
  1227→      case 'recent':
  1228→        return new Date(b.createdAt) - new Date(a.createdAt);
  1229→      case 'views':
  1230→        return (b.views || 0) - (a.views || 0);
  1231→      case 'photos':
  1232→        return b.gallery.length - a.gallery.length;
  1233→      case 'links':
  1234→        return b.websites.length - a.websites.length;
  1235→      default: // 'name'
  1236→        return a.name.localeCompare(b.name);
  1237→    }
  1238→  });
  1239→}

  1607→/**
  1608→ * Move a website up or down in the websites array and update the form display
  1609→ * @param {number} index - Current index of the website in the array
  1610→ * @param {string} direction - 'up' or 'down'
  1611→ */
  1612→function moveWebsite(index, direction) {
  1613→  // Get current editing actress from the actresses array
  1614→  if (!editingSlug) {
  1615→    showToast('Error: No actress is being edited', 'error');
  1616→    return;
  1617→  }
  1618→  
  1619→  const actress = actresses.find(a => a.slug === editingSlug);
  1620→  if (!actress || !actress.websites) {
  1621→    showToast('Error: Could not find websites to reorder', 'error');
  1622→    return;
  1623→  }
  1624→  
  1625→  // Make sure index is within bounds
  1626→  if (!Number.isInteger(index) || index < 0 || index >= actress.websites.length) {
  1627→    console.warn('Invalid index:', index, 'Total websites:', actress.websites.length);
  1628→    showToast('Error: Invalid website index', 'error');
  1629→    return;
  1630→  }
  1631→  
  1632→  // Determine the swap target index
  1633→  let targetIndex = -1;
  1634→  if (direction === 'up' && index > 0) {
  1635→    targetIndex = index - 1;
  1636→  } else if (direction === 'down' && index < actress.websites.length - 1) {
  1637→    targetIndex = index + 1;
  1638→  } else {
  1639→    showToast(`Cannot move ${direction} - already at ${direction === 'up' ? 'top' : 'bottom'}`, 'info');
  1640→    return;
  1641→  }
  1642→  
  1643→  // Swap the websites in the array
  1644→  [actress.websites[index], actress.websites[targetIndex]] = 
  1645→  [actress.websites[targetIndex], actress.websites[index]];
  1646→  
  1647→  // Re-populate the form to reflect the new order
  1648→  populateFormForEditing(actress);
  1649→  
  1650→  console.log(`Website moved ${direction}: old index ${index}, new index ${targetIndex}`);
  1651→}
